1class Solution {
2    class State{
3        int x, y, parity;
4        long cost;
5        State(int x, int y, int parity,long cost){
6            this.x = x;
7            this.y = y;
8            this.parity = parity;
9            this.cost = cost;
10        }
11    }
12    public long minCost(int m, int n, int[][] penalty) {
13        long INF = Long.MAX_VALUE/4;
14        long[][][] dist = new long[m][n][2];
15
16        for(int i = 0; i<m; i++){
17            for(int j = 0; j<n; j++){
18                Arrays.fill(dist[i][j], INF);
19            }
20        }
21
22        PriorityQueue<State> pq = new PriorityQueue<>((a, b)->Long.compare(a.cost, b.cost));
23        dist[0][0][0] = 1;
24        pq.offer(new State(0,0,0,1));
25
26        int[] dx = {-1, 1, 0, 0};
27        int[] dy = {0, 0, -1, 1};
28
29        while(!pq.isEmpty()){
30            State cur = pq.poll();
31            if(cur.cost!=dist[cur.x][cur.y][cur.parity])continue;
32
33            if(cur.x == m-1 && cur.y == n-1)return cur.cost;
34
35            long waitCost = cur.cost + penalty[cur.x][cur.y];
36
37            if(waitCost < dist[cur.x][cur.y][cur.parity^1]){
38                dist[cur.x][cur.y][cur.parity^1] = waitCost;
39                pq.offer(new State(cur.x, cur.y, cur.parity^1, waitCost));
40            }
41
42            for(int k = 0; k<4; k++){
43                int nx = cur.x+dx[k];
44                int ny = cur.y+dy[k];
45
46                if(nx<0 || nx>=m || ny<0 || ny>=n)continue;
47
48                long add = (long)(nx+1)*(ny+1);
49
50                boolean followsRule;
51
52                if(cur.parity == 0){
53                    followsRule = (dx[k] == 1 && dy[k] == 0)|| (dx[k] == 0 && dy[k] == 1);
54                }else {
55                    followsRule = (dx[k] == -1 && dy[k] == 0)|| (dx[k] == 0 && dy[k] == -1);
56                }
57
58                if(!followsRule)add+=penalty[cur.x][cur.y];
59
60                long newCost = cur.cost+add;
61
62                if(newCost<dist[nx][ny][cur.parity^1]){
63                    dist[nx][ny][cur.parity^1] = newCost;
64                    pq.offer(new State(nx, ny, cur.parity^1, newCost));
65                }
66            }
67        }
68        return -1;
69    }
70}
